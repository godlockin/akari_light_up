  // 求解策略（按复杂度排序）
  const strategies = [
    '满数推理',    // 数字4周围必全是灯，角上2必全是灯
    '差一推理',    // 数字3周围有3个白格，则全是灯
    '唯一照亮',    // 某白格只能被一个位置照亮，该位置必是灯
    '排除推理',    // 放灯会导致冲突的位置必不能放灯
    '假设验证',    // 尝试放置，验证是否导致矛盾
  ];

  // 求解器返回
  interface SolveResult {
    solvable: boolean;      // 是否可解
    unique: boolean;        // 是否唯一解
    solution: Set<string>;  // 解(灯泡位置集合)
    steps: ReasoningStep[]; // 推理步骤(用于提示)
  }
